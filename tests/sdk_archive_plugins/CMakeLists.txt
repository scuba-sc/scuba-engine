set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(SCUBA_PLUGIN_SDK_ROOT "" CACHE PATH "Verified extracted combined plugin SDK")
if(NOT EXISTS "${SCUBA_PLUGIN_SDK_ROOT}/cmake/sco-plugin.cmake")
  message(FATAL_ERROR "Supply SCUBA_PLUGIN_SDK_ROOT from the verified plugin archive")
endif()
set(SCO_SDK_INCLUDE_DIR "${SCUBA_PLUGIN_SDK_ROOT}/include")
include("${SCUBA_PLUGIN_SDK_ROOT}/cmake/sco-plugin.cmake")
get_filename_component(engine_root "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
function(archive_plugin id)
  set(folder "${engine_root}/plugins/${id}")
  set(sources "")
  foreach(file IN LISTS ARGN)
    list(APPEND sources "${folder}/src/${file}")
  endforeach()
  sco_add_plugin(${id} SOURCES ${sources} INI "${folder}/plugin.ini")
  target_include_directories(${id} PRIVATE "${folder}/src")
  set_target_properties(${id} PROPERTIES LIBRARY_OUTPUT_DIRECTORY "$<1:${CMAKE_BINARY_DIR}/runtime/${id}>"
    RUNTIME_OUTPUT_DIRECTORY "$<1:${CMAKE_BINARY_DIR}/runtime/${id}>")
  configure_file("${folder}/plugin.ini" "${CMAKE_BINARY_DIR}/runtime/${id}/plugin.ini" COPYONLY)
  foreach(mode full short)
    set(extra "")
    if(mode STREQUAL "short")
      set(extra --short-tables)
    endif()
    add_test(NAME archive_${id}_${mode}
      COMMAND sco-plugin-check "${CMAKE_BINARY_DIR}/runtime/${id}" ${extra})
  endforeach()
endfunction()
enable_testing()
add_executable(sco-plugin-check "${SCUBA_PLUGIN_SDK_ROOT}/tools/sco-plugin-check.c")
target_include_directories(sco-plugin-check PRIVATE "${SCO_SDK_INCLUDE_DIR}")
set_target_properties(sco-plugin-check PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON

if(MSVC)

  target_compile_options(sco-plugin-check PRIVATE -Wall -Wextra -Werror)
endif()
target_link_libraries(sco-plugin-check PRIVATE ${CMAKE_DL_LIBS})
