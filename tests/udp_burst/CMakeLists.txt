cmake_minimum_required(VERSION 3.21)
project(scuba_udp_burst_regression LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
# Explicit local inputs keep this diagnostic separate from the released profile.
if(NOT SCUBA_SDK_ARCHIVE_DIR OR NOT SCUBA_SDK_HOST_SOURCE_SHA256)
  message(FATAL_ERROR "Supply the local SDK archive directory and independently verified SHA256")
endif()
include("${CMAKE_CURRENT_SOURCE_DIR}/../../cmake/ScubaSdk.cmake")
add_executable(test_udp_burst main.cpp)
target_link_libraries(test_udp_burst PRIVATE sco_net_service)
if(MSVC)
  target_compile_options(test_udp_burst PRIVATE /W4 /WX)
else()
  target_compile_options(test_udp_burst PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_test(NAME sdk_udp_receive_burst COMMAND test_udp_burst)
set_tests_properties(sdk_udp_receive_burst PROPERTIES TIMEOUT 5)
